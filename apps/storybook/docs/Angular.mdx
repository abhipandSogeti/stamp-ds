# Using Stamp DS in Angular

`@stamp-ds/angular` provides one Angular **directive** per Stamp component, bundled into
`StampModule` for easy import. Directives (not components) are used because the Web
Component handles all rendering — Angular only needs to type-check inputs and bridge outputs.

## One-time setup

### Step 1 — Load the token CSS

Add the token stylesheet to `angular.json` so it's bundled and loaded globally:

```json
// angular.json
{
  "projects": {
    "your-app": {
      "architect": {
        "build": {
          "options": {
            "styles": [
              "node_modules/@stamp-ds/tokens/dist/css/tokens.css",
              "src/styles.css"
            ]
          }
        }
      }
    }
  }
}
```

Alternatively, import it in your global `src/styles.css`:

```css
/* src/styles.css */
@import 'node_modules/@stamp-ds/tokens/dist/css/tokens.css';
```

### Step 2 — Register custom elements and import StampModule

#### NgModule-based apps (Angular < 17 default)

```ts
// src/app/app.module.ts
import { NgModule, CUSTOM_ELEMENTS_SCHEMA } from '@angular/core';
import { BrowserModule } from '@angular/platform-browser';
import { StampModule } from '@stamp-ds/angular';
import '@stamp-ds/components'; // registers all <stamp-*> custom elements globally

import { AppComponent } from './app.component';

@NgModule({
  declarations: [AppComponent],
  imports: [
    BrowserModule,
    StampModule,   // ← provides all Stamp directives
  ],
  schemas: [
    CUSTOM_ELEMENTS_SCHEMA, // ← required: tells Angular to accept <stamp-*> tags
  ],
  bootstrap: [AppComponent],
})
export class AppModule {}
```

> **Why `CUSTOM_ELEMENTS_SCHEMA`?** By default Angular throws a template error when it
> encounters an element with a hyphen (`<stamp-button>`) that it doesn't recognise. This
> schema tells the compiler to allow it. You must add it wherever a module or standalone
> component uses Stamp elements in its template.

#### Standalone components (Angular 17+ default)

```ts
// src/app/app.component.ts
import { Component, CUSTOM_ELEMENTS_SCHEMA } from '@angular/core';
import { StampModule } from '@stamp-ds/angular';
import '@stamp-ds/components';

@Component({
  standalone: true,
  selector: 'app-root',
  imports: [StampModule],
  schemas: [CUSTOM_ELEMENTS_SCHEMA], // ← required per-component when standalone
  template: `
    <stamp-button variant="solid" (stampClick)="save()">Save</stamp-button>
  `,
})
export class AppComponent {
  save() { /* ... */ }
}
```

> **Prefer per-component imports over StampModule** in large standalone apps.
> Import only what you use to keep the template compiler scope minimal:
>
> ```ts
> import { ButtonDirective, InputDirective } from '@stamp-ds/angular';
>
> @Component({
>   standalone: true,
>   imports: [ButtonDirective, InputDirective],
>   schemas: [CUSTOM_ELEMENTS_SCHEMA],
>   // ...
> })
> ```

**Component class:**

```ts
// button.component.ts
import { Component } from '@angular/core';

@Component({ /* ... */ })
export class MyComponent {
  isLoading = false;
  isDisabled = false;

  async onSave(event: CustomEvent<{ originalEvent: MouseEvent }>) {
    this.isLoading = true;
    try {
      await saveData();
    } finally {
      this.isLoading = false;
    }
  }
}
```

**Angular Reactive Forms integration:**

Stamp Input is not a native `ControlValueAccessor` (that would require a separate adapter).
The recommended pattern is to sync the form control value manually:

```ts
import { Component, OnInit } from '@angular/core';
import { FormBuilder, FormGroup, Validators, ReactiveFormsModule } from '@angular/forms';
import { StampModule } from '@stamp-ds/angular';
import { CUSTOM_ELEMENTS_SCHEMA } from '@angular/core';

@Component({
  standalone: true,
  imports: [ReactiveFormsModule, StampModule],
  schemas: [CUSTOM_ELEMENTS_SCHEMA],
  template: `
    <form [formGroup]="form" (ngSubmit)="onSubmit()">
      <stamp-input
        label="Email"
        type="email"
        [value]="form.get('email')?.value ?? ''"
        [error]="emailError"
        (stampInput)="form.get('email')?.setValue($event.detail.value)"
      ></stamp-input>
      <stamp-button type="submit" variant="solid">Submit</stamp-button>
    </form>
  `,
})
export class LoginFormComponent implements OnInit {
  form!: FormGroup;

  get emailError(): string {
    const ctrl = this.form.get('email');
    if (ctrl?.invalid && ctrl.touched) return 'Enter a valid email';
    return '';
  }

  constructor(private fb: FormBuilder) {}

  ngOnInit() {
    this.form = this.fb.group({
      email: ['', [Validators.required, Validators.email]],
    });
  }

  onSubmit() {
    this.form.markAllAsTouched();
    if (this.form.valid) { /* submit */ }
  }
}
```

**Select-all pattern (indeterminate):**

```ts
@Component({
  template: `
    <stamp-checkbox
      label="Select all"
      [checked]="allSelected"
      [indeterminate]="someSelected"
      (stampChange)="toggleAll()"
    ></stamp-checkbox>
    <stamp-checkbox
      *ngFor="let item of items"
      [label]="item.name"
      [checked]="item.selected"
      (stampChange)="toggle(item)"
    ></stamp-checkbox>
  `,
})
export class SelectAllComponent {
  items = [
    { name: 'Apples', selected: false },
    { name: 'Bananas', selected: false },
    { name: 'Cherries', selected: false },
  ];

  get allSelected() { return this.items.every((i) => i.selected); }
  get someSelected() { return this.items.some((i) => i.selected) && !this.allSelected; }

  toggleAll() {
    const next = !this.allSelected;
    this.items = this.items.map((i) => ({ ...i, selected: next }));
  }

  toggle(item: { name: string; selected: boolean }) {
    item.selected = !item.selected;
  }
}
```

## Change detection

Stamp components are Lit Web Components — they manage their own rendering outside
Angular's change detection tree. This has two practical implications:

**Property bindings work without `ChangeDetectionStrategy.OnPush`:**

```html
<!-- This updates correctly even with OnPush -->
<stamp-button [loading]="isLoading">Save</stamp-button>
```

**Output events trigger Angular change detection automatically:**

Angular's event binding `(stampClick)="..."` runs inside Angular's zone, so state
updates inside the handler are reflected in the template immediately — no
`markForCheck()` needed.

**When using `ChangeDetectionStrategy.OnPush` and reading output values:**

```ts
@Component({
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class MyComponent {
  value = '';

  constructor(private cd: ChangeDetectorRef) {}

  onInput(e: CustomEvent<{ value: string }>) {
    this.value = e.detail.value;
    // Angular's (stampInput) binding handles markForCheck() automatically
    // Only call cd.markForCheck() if you update state outside event handlers
  }
}
```

## Angular Universal (SSR)

Stamp components use browser APIs (`customElements.define`, shadow DOM). If you render
server-side with Angular Universal, guard the registration:

```ts
// app.module.ts or main.ts
import { isPlatformBrowser } from '@angular/common';
import { PLATFORM_ID, inject } from '@angular/core';

// In a service or APP_INITIALIZER
const platformId = inject(PLATFORM_ID);
if (isPlatformBrowser(platformId)) {
  import('@stamp-ds/components'); // dynamic import — skipped on server
}
```

Token CSS is plain CSS and is safe to load in any environment.

## Dark mode

Toggle the token stylesheet link at runtime. The simplest approach:

```ts
// src/app/services/theme.service.ts
import { Injectable } from '@angular/core';

@Injectable({ providedIn: 'root' })
export class ThemeService {
  private isDark = false;

  toggle() {
    this.isDark = !this.isDark;
    const link = document.getElementById('stamp-tokens') as HTMLLinkElement;
    if (!link) return;
    link.href = this.isDark
      ? 'assets/tokens.dark.css'   // copy from node_modules in angular.json assets
      : 'assets/tokens.css';
  }
}
```

In `angular.json`, copy the token files to `src/assets/`:

```json
"assets": [
  "src/favicon.ico",
  "src/assets",
  {
    "glob": "tokens*.css",
    "input": "node_modules/@stamp-ds/tokens/dist/css",
    "output": "assets"
  }
]
```

In `index.html`:

```html
<link id="stamp-tokens" rel="stylesheet" href="assets/tokens.css" />
```

See [Theming](?path=/docs/stamp-ds-theming--docs) for the full guide.

## Theming / token overrides

Override CSS custom properties in your `styles.css` after the token import:

```css
/* src/styles.css */
:root {
  /* Brand colour override */
  --stamp-color-brand-primary: #7C3AED;

  /* Global radius change */
  --stamp-button-radius: 9999px;
}
```

Per-component override via a CSS class:

```css
/* Only affect buttons in the checkout footer */
.checkout-footer stamp-button {
  --stamp-button-bg: #7C3AED;
  --stamp-button-radius: 9999px;
}
```
