# Theming

Stamp's theming model is CSS-native: design tokens are CSS custom properties that cascade
from the document root into every component's Shadow DOM. Switching themes means swapping
one stylesheet — no JavaScript theme providers, no React context, no Angular services.

---

No per-component changes are needed.

## Dark Mode

### Option 1: HTML attribute toggle (recommended)

The cleanest approach — keep both stylesheets in `<head>` and toggle an attribute:

```html
<head>
  <link id="stamp-theme" rel="stylesheet" href="@stamp-ds/tokens/dist/css/tokens.css" />
</head>
```

```js
function setTheme(scheme) {
  const link = document.getElementById('stamp-theme');
  const base = 'node_modules/@stamp-ds/tokens/dist/css/';
  link.href = scheme === 'dark' ? `${base}tokens.dark.css` : `${base}tokens.css`;
  document.documentElement.setAttribute('data-scheme', scheme);
}

// Toggle
setTheme('dark');
setTheme('light');
```

Use `data-scheme` for any custom CSS you want to scope to a theme:

```css
[data-scheme='dark'] .my-sidebar {
  backdrop-filter: blur(8px);
}
```

### Option 2: Follow the OS preference automatically

```js
const mq = window.matchMedia('(prefers-color-scheme: dark)');

function applySystemTheme() {
  setTheme(mq.matches ? 'dark' : 'light');
}

applySystemTheme();               // apply on load
mq.addEventListener('change', applySystemTheme);  // respond to OS changes
```

### Option 3: Load the correct file server-side

For SSR apps, emit the correct `<link>` based on a cookie or `Accept-CH` hint:

```ts
// Next.js: pages/_document.tsx
import Document, { Html, Head, Main, NextScript } from 'next/document';

export default class MyDocument extends Document {
  render() {
    // Read from cookie in getInitialProps for real implementation
    const scheme = 'light';
    const tokenFile = scheme === 'dark' ? 'tokens.dark.css' : 'tokens.css';

    return (
      <Html data-scheme={scheme}>
        <Head>
          <link
            rel="stylesheet"
            href={`/_next/static/css/${tokenFile}`}
          />
        </Head>
        <body>
          <Main />
          <NextScript />
        </body>
      </Html>
    );
  }
}
```

### React integration

```tsx
// useTheme.ts
import { useState, useEffect } from 'react';

type Scheme = 'light' | 'dark';

export function useTheme() {
  const [scheme, setScheme] = useState<Scheme>('light');

  useEffect(() => {
    const link = document.getElementById('stamp-theme') as HTMLLinkElement;
    const base = '/node_modules/@stamp-ds/tokens/dist/css/';
    link.href = scheme === 'dark' ? `${base}tokens.dark.css` : `${base}tokens.css`;
    document.documentElement.setAttribute('data-scheme', scheme);
  }, [scheme]);

  return { scheme, setScheme };
}

// ThemeToggle.tsx
import { useTheme } from './useTheme';
import { Button } from '@stamp-ds/react';

export function ThemeToggle() {
  const { scheme, setScheme } = useTheme();
  return (
    <Button
      variant="ghost"
      onStampClick={() => setScheme(scheme === 'light' ? 'dark' : 'light')}
    >
      {scheme === 'light' ? 'Dark' : 'Light'}
    </Button>
  );
}
```

### Angular integration

```ts
// theme.service.ts
import { Injectable, Inject } from '@angular/core';
import { DOCUMENT } from '@angular/common';

@Injectable({ providedIn: 'root' })
export class ThemeService {
  private scheme: 'light' | 'dark' = 'light';

  constructor(@Inject(DOCUMENT) private doc: Document) {}

  toggle() {
    this.scheme = this.scheme === 'light' ? 'dark' : 'light';
    this.apply();
  }

  private apply() {
    const link = this.doc.getElementById('stamp-theme') as HTMLLinkElement;
    const base = 'node_modules/@stamp-ds/tokens/dist/css/';
    link.href = `${base}tokens${this.scheme === 'dark' ? '.dark' : ''}.css`;
    this.doc.documentElement.setAttribute('data-scheme', this.scheme);
  }
}
```

## Custom Brand Theming

Override semantic tokens to apply a custom brand palette. Place overrides *after*
loading `tokens.css` so they win the cascade:

```css
/* brand.css */
:root {
  /* Primary brand colour */
  --stamp-color-brand-primary:       #7C3AED;  /* violet-600 */
  --stamp-color-brand-primary-hover: #6D28D9;  /* violet-700 */
  --stamp-color-brand-primary-text:  #FFFFFF;

  /* Secondary accent */
  --stamp-color-brand-secondary:     #059669;  /* emerald-600 */

  /* Shape language — square corners */
  --stamp-radius-sm: 0;
  --stamp-radius-md: 2px;
  --stamp-radius-lg: 4px;
  --stamp-radius-xl: 6px;

  /* Typography */
  --stamp-font-family-sans: 'Figtree', system-ui, sans-serif;
}
```

```html
<head>
  <link rel="stylesheet" href="@stamp-ds/tokens/dist/css/tokens.css" />
  <link rel="stylesheet" href="brand.css" />  <!-- overrides after -->
</head>
```

### Dark mode variant of your brand

Provide dark-mode overrides scoped to `[data-scheme='dark']`:

```css
/* brand.css */
:root {
  --stamp-color-brand-primary: #7C3AED;
}

[data-scheme='dark'] {
  --stamp-color-brand-primary: #A78BFA;  /* lighter violet for dark backgrounds */
}
```

## Per-Component Overrides

Override component-tier tokens to restyle a single component type without touching
the global palette:

```css
/* Make all buttons in the marketing nav pill-shaped */
.marketing-nav stamp-button {
  --stamp-button-radius: 9999px;
  --stamp-button-padding-x: 1.5rem;
}

/* Wider inputs in the checkout flow */
.checkout-form stamp-input {
  --stamp-input-min-width: 360px;
  --stamp-input-border-width: 2px;
}
```

To target a single instance, use an inline style or a unique class on the element:

```html
<stamp-button
  style="--stamp-button-bg: #7C3AED; --stamp-button-bg-hover: #6D28D9;"
  variant="solid"
>
  Special action
</stamp-button>
```
