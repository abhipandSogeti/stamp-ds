# Getting Started

Choose your framework below. All frameworks consume the same Web Components runtime —
the framework-specific packages are thin wrappers that add typed props and idiomatic event APIs.

---

## Prerequisites

- Node.js 18 or later
- npm 9 / pnpm 8 / yarn 3 or later
- A bundler that understands ES modules (Vite, Webpack 5, Rollup, esbuild)

---

## Vanilla HTML

The quickest path — no framework, no build step required.

### Via CDN (prototyping only)

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <!-- 1. Load design tokens (CSS custom properties) -->
  <link rel="stylesheet"
        href="https://unpkg.com/@stamp-ds/tokens/dist/css/tokens.css" />
</head>
<body>

  <!-- 2. Use components as standard HTML elements -->
  <stamp-button variant="solid">Save changes</stamp-button>
  <stamp-input label="Email" placeholder="you@example.com"></stamp-input>

  <!-- 3. Register components (auto-defines all custom elements) -->
  <script type="module"
          src="https://unpkg.com/@stamp-ds/components/dist/index.js"></script>
</body>
</html>
```

### Via npm + Vite

```bash
npm install @stamp-ds/tokens @stamp-ds/components
```

```js
// main.js
import '@stamp-ds/tokens/dist/css/tokens.css';
import '@stamp-ds/components'; // registers all <stamp-*> elements
```

```html
<!-- index.html -->
<stamp-button variant="solid">Save</stamp-button>
<stamp-input label="Email" placeholder="you@example.com" required></stamp-input>
<stamp-checkbox label="Accept terms"></stamp-checkbox>
```

### Listening to events

Stamp components dispatch standard DOM `CustomEvent`s on the element itself.

```js
const btn = document.querySelector('stamp-button');
btn.addEventListener('stamp-click', (e) => {
  console.log('clicked', e.detail);
});

const input = document.querySelector('stamp-input');
input.addEventListener('stamp-input', (e) => {
  console.log('value:', e.detail.value);
});

const checkbox = document.querySelector('stamp-checkbox');
checkbox.addEventListener('stamp-change', (e) => {
  console.log('checked:', e.detail.checked);
});
```

## React

Stamp ships `@stamp-ds/react` — a set of React components generated by
`@lit-labs/react` that wrap each Web Component with typed props and idiomatic
`on*` callback props.

### Install

```bash
npm install @stamp-ds/tokens @stamp-ds/components @stamp-ds/react
```

### Basic usage

```tsx
// index.tsx (or main.tsx)
import '@stamp-ds/tokens/dist/css/tokens.css';  // load CSS custom properties once

// In any component file:
import { Button, Input, Checkbox, Badge, Spinner } from '@stamp-ds/react';

export function LoginForm() {
  return (
    <form onSubmit={(e) => e.preventDefault()}>
      <Input
        label="Email address"
        type="email"
        placeholder="you@example.com"
        required
        onStampInput={(e) => console.log(e.detail.value)}
      />
      <Input
        label="Password"
        type="password"
        placeholder="••••••••"
        required
      />
      <Checkbox
        label="Remember me"
        onStampChange={(e) => console.log(e.detail.checked)}
      />
      <Button
        variant="solid"
        type="submit"
        onStampClick={() => console.log('submit')}
      >
        Sign in
      </Button>
    </form>
  );
}
```

### TypeScript types

All props are fully typed. Your IDE will autocomplete variants, sizes, and event payloads:

```tsx
import type { StampClickEvent } from '@stamp-ds/react';

function DeleteButton() {
  const handleClick = (e: StampClickEvent) => {
    console.log(e.detail); // typed CustomEventDetail
  };

  return (
    <Button variant="danger" size="md" onStampClick={handleClick}>
      Delete account
    </Button>
  );
}
```

### Refs

React wrappers forward refs to the underlying Web Component element:

```tsx
import { useRef, useEffect } from 'react';
import { Input } from '@stamp-ds/react';

function AutoFocusInput() {
  const ref = useRef<HTMLElement>(null);

  useEffect(() => {
    ref.current?.focus();
  }, []);

  return <Input ref={ref} label="Name" />;
}
```

### Next.js (App Router)

Web Components run in the browser only. Mark any file that imports Stamp components
with the `'use client'` directive:

```tsx
'use client';

import { Button } from '@stamp-ds/react';

export function SubmitButton() {
  return <Button variant="solid">Submit</Button>;
}
```

No SSR polyfills are needed for the token CSS — it's a plain stylesheet.

### Next.js (Pages Router)

Import the token CSS in `_app.tsx`:

```tsx
// pages/_app.tsx
import '@stamp-ds/tokens/dist/css/tokens.css';
import type { AppProps } from 'next/app';

export default function App({ Component, pageProps }: AppProps) {
  return <Component {...pageProps} />;
}
```

Dynamic-import components that use Web Components if you hit SSR issues:

```tsx
import dynamic from 'next/dynamic';

const Button = dynamic(
  () => import('@stamp-ds/react').then((m) => m.Button),
  { ssr: false }
);
```

## Angular

`@stamp-ds/angular` provides one Angular directive per Stamp component, bundled
into `StampModule` for easy import.

### Install

```bash
npm install @stamp-ds/tokens @stamp-ds/components @stamp-ds/angular
```

### Enable custom elements in Angular

Angular requires `CUSTOM_ELEMENTS_SCHEMA` when using Web Components:

```ts
// app.module.ts
import { NgModule, CUSTOM_ELEMENTS_SCHEMA } from '@angular/core';
import { BrowserModule } from '@angular/platform-browser';
import { StampModule } from '@stamp-ds/angular';
import '@stamp-ds/components'; // registers all <stamp-*> custom elements

import { AppComponent } from './app.component';

@NgModule({
  declarations: [AppComponent],
  imports: [BrowserModule, StampModule],
  schemas: [CUSTOM_ELEMENTS_SCHEMA],
  bootstrap: [AppComponent],
})
export class AppModule {}
```

Load token CSS globally in `angular.json`:

```json
{
  "projects": {
    "your-app": {
      "architect": {
        "build": {
          "options": {
            "styles": [
              "node_modules/@stamp-ds/tokens/dist/css/tokens.css"
            ]
          }
        }
      }
    }
  }
}
```

### Template usage

```html
<!-- login.component.html -->

<stamp-input
  label="Email address"
  type="email"
  [placeholder]="emailPlaceholder"
  [required]="true"
  (stampInput)="onEmailInput($event)"
></stamp-input>

<stamp-input
  label="Password"
  type="password"
  placeholder="••••••••"
  [error]="passwordError"
></stamp-input>

<stamp-checkbox
  label="Remember me"
  [checked]="rememberMe"
  (stampChange)="onRememberMe($event)"
></stamp-checkbox>

<stamp-button
  variant="solid"
  [loading]="isSubmitting"
  (stampClick)="onSubmit()"
>
  Sign in
</stamp-button>
```

```ts
// login.component.ts
import { Component } from '@angular/core';

@Component({
  selector: 'app-login',
  templateUrl: './login.component.html',
})
export class LoginComponent {
  emailPlaceholder = 'you@example.com';
  passwordError = '';
  rememberMe = false;
  isSubmitting = false;

  onEmailInput(event: CustomEvent<{ value: string }>) {
    console.log('email:', event.detail.value);
  }

  onRememberMe(event: CustomEvent<{ checked: boolean }>) {
    this.rememberMe = event.detail.checked;
  }

  async onSubmit() {
    this.isSubmitting = true;
    // ... submit logic
    this.isSubmitting = false;
  }
}
```

## Vite configuration

No special config is needed. Vite handles ES modules and CSS imports natively.

If you get `[vite] Failed to resolve import` errors for the token CSS, add an alias:

```ts
// vite.config.ts
import { defineConfig } from 'vite';

export default defineConfig({
  resolve: {
    alias: {
      '@stamp-ds/tokens/css': '@stamp-ds/tokens/dist/css/tokens.css',
    },
  },
});
```

## Webpack 5 configuration

```js
// webpack.config.js
module.exports = {
  module: {
    rules: [
      {
        test: /\.css$/,
        use: ['style-loader', 'css-loader'],
      },
    ],
  },
  experiments: {
    outputModule: true, // required for ES module output
  },
};
```
